      \begin{block}{Problem: Evaluation Crisis}
          Generative music evaluation is \textbf{methodologically fragmented}, relies on \textbf{ambiguous validation objectives}, and builds on \textbf{systematically misaligned metrics} \cite{lerch_survey_2025}. %This has led to a crisis in the field, where the following failures are observed:
        
        \begin{itemize}
          \bigskip
          \item \textbf{Validity failure:} metrics poorly capture musical qualities.
              % \begin{itemize}
              %   \item Utilized metrics are \textit{unreliable proxies} and fail to measure musical qualities.
              % \end{itemize}
          \bigskip
          \item \textbf{Leaderboard failure:}  single scores encourage optimization artifacts and obscure musical multi-dimensionality.
              % \begin{itemize}
              %   \item Single-score optimization leads to \textit{overfitting} and is \textit{insufficient for assessing a multi-dimensional cultural artifact}.
              % \end{itemize}
          \bigskip
          \item \textbf{Comparability failure:} different evaluation pipelines yield incomparable results.
              % \begin{itemize}
              %   \item \textit{Different metrics, feature representations, and reference data} prevent meaningful comparisons across models.
              % \end{itemize}
          \bigskip
          \item \textbf{Reproducibility failure:} proprietary data and tools limit verification.
            %  \begin{itemize}
            %     \item \textit{Closed-source representations and evaluation scripts} and proprietary data hinder reproducibility.
            %   \end{itemize}
          \bigskip
          \item \textbf{Governance failure:} no shared standards or process for evolution.
              % \begin{itemize}
              %   \item \textit{Ad-hoc, fragmented evaluation protocols} indicate lack of standardization.
              % \end{itemize}
        \end{itemize}

      \end{block}

      \begin{block}{Core Ideas}
        \begin{itemize}
          \bigskip
          \item \textbf{Community governance:} \textit{community-maintained evolving benchmarks} and evaluation standards.
              % \begin{itemize}
              %   \item coordinated by the community for the community. 
              %   \item metrics and evaluation protocols have to \textit{evolve with the field}.
              % \end{itemize}
          \bigskip
          \item \textbf{Multi-dimensional assessment:} capture many \textit{relevant dimensions} instead one single ambiguous score.
              % \begin{itemize}
              %   \item acknowledging the \textit{multi-dimensional nature of music}.
              % \end{itemize}
          \bigskip
          \item \textbf{Interpretability:} utilize \textit{interpretable metrics} and features.
              % \begin{itemize}
              %   \item utilizing \textit{interpretable metrics} and features where possible.
              % \end{itemize}
          \bigskip
          \item \textbf{Use-case adaptability:} tailor curated \textit{reference benchmark profiles (RBPs)} to specific musical contexts.
              % \begin{itemize}
              %   \item acknowledging \textit{variety of use cases} through curated reference benchmark profiles (RBPs).
              % \end{itemize}
           \bigskip
          \item \textbf{Modular, extensible architecture:} new metrics, features, and data can be \textit{integrated without redesign}.
            %  \begin{itemize}
            %     \item modular software architecture allows for easy \textit{integration of new features and evaluation methods}.
            %   \end{itemize}
          \bigskip
          \item \textbf{Reproducibility \& transparency:} versioned, documented \textit{open source} software with \textit{auditable logging}.
            %  \begin{itemize}
            %     \item versioned, documented \textit{open source} software with \textit{auditable logging}.
            %   \end{itemize}
        \end{itemize}

      \end{block}

      \vskip.3ex
      \begin{block}{Limitations}
        \begin{itemize}
            \item Missing \textbf{subjective evaluation protocols}
            \bigskip
            \item \textbf{Black-box evaluation}:
              \begin{itemize}
                  \item no evaluation of the underlying model architecture/capabilities
                  \item no distinction between human contributions and fully machine-generated output
              \end{itemize}
           \bigskip
            \item \textbf{Focus on output qualities}:
              \begin{itemize}
                \item no usability or user experience evaluation
                \item no evaluation other characteristics, such as explainability, bias, resource usage, ethical use of data, etc.
              \end{itemize}
        \end{itemize}
      \end{block}
